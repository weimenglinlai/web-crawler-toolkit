import time
import json
import sys
import re
import pandas as pd
import os
from selenium import webdriver
from selenium.webdriver.chrome.service import Service
from selenium.common import NoSuchElementException
from selenium.webdriver.common.by import By

# ==================== 配置区域 ====================
data_path = "datas"
csv_file = "gameDatas.csv"

# ==================== 函数定义 ====================
def init_csv():
    """初始化 CSV 文件"""
    if not os.path.exists(data_path):
        os.makedirs(data_path)
    df_header = pd.DataFrame(columns=["Name", "Price", "Rating"])
    df_header.to_csv(csv_file, encoding="utf_8_sig", index=False, mode="w")


def save_infos(name, price, rating):
    # 保存为 JSON
    with open(data_path + f"/{hash(name)}.json", "w", encoding="utf-8") as data_file:
        data = {"Name": name, "Price": price, "Rating": rating}
        json.dump(data, data_file, indent=4, ensure_ascii=False)

    # 追加到 CSV
    new_line = pd.DataFrame([{"Name": name, "Price": price, "Rating": rating}])
    new_line.to_csv(csv_file, encoding="utf_8_sig", index=False, mode="a", header=False)

def get_rating_fast(driver):
    try:
        # 方法1：查找包含评价的区块，直接取文本前50字符
        review_elem = driver.find_element(By.CLASS_NAME, "user_reviews_summary")
        if review_elem:
            text = review_elem.text
            if text:
                # 只取第一行或前50字符
                lines = text.strip().split('\n')
                for line in lines:
                    if '好评' in line or '差评' in line:
                        return line.strip()[:60]
    except:
        pass

    try:
        # 方法2：直接搜索页面中的评价关键词
        page_text = driver.find_element(By.TAG_NAME, "body").text
        # 匹配 "特别好评 (2,370)" 或 "褒贬不一 (114)" 等格式
        patterns = [
            r'([^、\n]{2,6}?(?:好评|差评)[^、\n]{0,4}?)\s*\([\d,]+\)',  # 匹配 "特别好评 (2,370)"
            r'(?:全部评测|总体评测)[：:]\s*\n?\s*([^\(\n]+)\s*\(',  # 匹配 "所有评测：\n特别好评 ("
        ]
        for pattern in patterns:
            match = re.search(pattern, page_text)
            if match:
                rating = match.group(1).strip()
                if rating:
                    return rating[:30]
    except:
        pass

    return "暂无评价"

def get_price_fast(driver):
    """快速获取价格"""
    try:
        # 直接尝试最常见的价格标签
        price_elem = driver.find_element(By.CLASS_NAME, "game_purchase_price")
        if price_elem and price_elem.text:
            return price_elem.text.strip()
    except:
        pass
    try:
        price_elem = driver.find_element(By.CLASS_NAME, "discount_final_price")
        if price_elem and price_elem.text:
            return price_elem.text.strip()
    except:
        pass

    return "价格未知"

# ==================== 主程序 ====================
def main():
    print("=" * 60)
    print("蒸汽平台游戏数据快速采集工具（高速版）")
    print("=" * 60)

    # ChromeDriver 路径
    chrome_driver_path = r"./data"

    if not os.path.exists(chrome_driver_path):
        print(f"\n错误：找不到 ChromeDriver！")
        print(f"路径: {chrome_driver_path}")
        input("按 Enter 退出...")
        return
    # 初始化
    init_csv()

    # 获取游戏列表
    print("\n[1/3] 正在获取游戏列表...")
    service = Service(chrome_driver_path)
    driver = webdriver.Chrome(service=service)
    driver.get("https://store.steamchina.com/search/?sort_by=Reviews_DESC")
    time.sleep(2)

    # 快速滚动（减少等待时间）
    print("  正在滚动加载...")
    for i in range(8):
        driver.execute_script("window.scrollTo(0, document.body.scrollHeight);")
        time.sleep(1)  # 减少到1秒
        print(f"    第{i + 1}次滚动")
    time.sleep(1)

    # 提取链接
    print("  正在提取链接...")
    game_links = []
    try:
        games = driver.find_element(By.ID, "search_resultsRows")
        for a in games.find_elements(By.TAG_NAME, "a"):
            href = a.get_attribute("href")
            if href and href.startswith("https://store.steamchina.com"):
                game_links.append(href)
        game_links = list(dict.fromkeys(game_links))
        print(f"  ✓ 找到 {len(game_links)} 个游戏")
    except Exception as e:
        print(f"  ✗ 提取失败: {e}")
        driver.quit()
        return
    driver.quit()

    # 询问采集数量
    print(f"\n[2/3] 设置采集数量")
    print(f"  当前共有 {len(game_links)} 个游戏")
    user_input = input(f"  要采集多少个？(直接回车=100，输入'all'=全部): ").strip()

    if user_input.lower() == 'all':
        limit = len(game_links)
    elif user_input.isdigit():
        limit = min(int(user_input), len(game_links))
    else:
        limit = min(100, len(game_links))
    print(f"  → 将采集前 {limit} 个游戏")

    # 快速采集详情
    print(f"\n[3/3] 开始快速采集（预计 {limit * 2} 秒）...")
    service2 = Service(chrome_driver_path)
    detail_driver = webdriver.Chrome(service=service2)

    success = 0
    fail = 0

    for idx, url in enumerate(game_links[:limit], 1):
        print(f"\n[{idx}/{limit}] ", end="")

        try:
            detail_driver.get(url)
            time.sleep(0.8)  # 减少到0.8秒

            # 获取名称
            try:
                name = detail_driver.find_element(By.CLASS_NAME, "apphub_AppName").text
            except:
                name = "未知"
            # 获取价格
            price = get_price_fast(detail_driver)
            # 获取好评率
            rating = get_rating_fast(detail_driver)
            # 保存
            save_infos(name, price, rating)
            success += 1

            # 简洁输出
            print(f"✓ {name[:25]:25} | {price:12} | {rating[:20]}")

        except Exception as e:
            fail += 1
            print(f"✗ 失败: {str(e)[:50]}")

    detail_driver.quit()

    # 完成统计
    print("\n" + "=" * 60)
    print("采集完成！")
    print(f"  成功: {success} 条")
    print(f"  失败: {fail} 条")
    print(f"  文件: {csv_file}")
    print("=" * 60)

if __name__ == "__main__":
    try:
        main()
    except KeyboardInterrupt:
        print("\n\n用户中断")
    except Exception as e:
        print(f"\n错误: {e}")
    finally:
        input("\n按 Enter 退出...")
